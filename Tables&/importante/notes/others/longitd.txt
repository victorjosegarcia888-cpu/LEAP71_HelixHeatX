✅ 3. SIMULADOR DE CÁMARA + TOBERA (C#)
(modelo físico basado en Sutton + NASA SP‑8120)

csharp
public class ChamberNozzleSim
{
    public double Pc, Tc, At, Ae, gamma, R;

    public ChamberNozzleSim(double Pc, double Tc, double At, double Ae, double gamma, double R)
    {
        this.Pc = Pc; this.Tc = Tc; this.At = At; this.Ae = Ae;
        this.gamma = gamma; this.R = R;
    }

    public double CStar()
    {
        return (Pc * At) / MassFlow();
    }

    public double MassFlow()
    {
        double term = Math.Sqrt(gamma / R * Tc);
        return Pc * At / term;
    }

    public double ExitVelocity()
    {
        double pr = 1e5 / Pc; // presión ambiente 1 bar
        double term = (2 * gamma / (gamma - 1)) * R * Tc *
                      (1 - Math.Pow(pr, (gamma - 1) / gamma));
        return Math.Sqrt(term);
    }

    public double Thrust()
    {
        double ve = ExitVelocity();
        double mdot = MassFlow();
        return mdot * ve + (1e5 - Pc) * Ae;
    }
}
✅ 4. SIMULADOR DE PLASMA (C#) — Basado en Chen
csharp
public class PlasmaSim
{
    const double epsilon0 = 8.854e-12;
    const double k = 1.38e-23;
    const double echarge = 1.602e-19;
    const double me = 9.11e-31;

    public double DebyeLength(double Te, double ne)
    {
        return Math.Sqrt(epsilon0 * k * Te / (ne * echarge * echarge));
    }

    public double PlasmaFrequency(double ne)
    {
        return Math.Sqrt(ne * echarge * echarge / (epsilon0 * me));
    }

    public double Conductivity(double ne, double tau)
    {
        return ne * echarge * echarge * tau / me;
    }
}

# Simulador de inestabilidad acústica (Sutton)

[Entrada]
D = diámetro cámara
L = longitud cámara
Tc = temperatura combustión
gamma = 1.2
R = 350 J/kgK

[Velocidad del sonido]
a = sqrt(gamma * R * Tc)

[Frecuencias]
Longitudinal:
  f_L = a / (2L)

Radial:
  f_R = 1.84 * a / (πD)

Tangencial:
  f_T = a / (D)

[Detección de inestabilidad]
Si presión oscila > 5% → inestabilidad leve
Si presión oscila > 20% → inestabilidad severa
Si f coincide con modo natural → riesgo alto

[Mitigación]
- Añadir baffles (NASA)
- Cambiar patrón de inyección
- Añadir cavidades Helmholtz
- Ajustar O/F
